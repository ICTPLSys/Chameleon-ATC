cmake_minimum_required(VERSION 3.16)
project(fig9_generator LANGUAGES C CXX)
include(CTest)
get_filename_component(MCGEN_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../memcached-trace-generator" ABSOLUTE)
add_subdirectory("${MCGEN_SOURCE}" upstream)
get_target_property(MCGEN_SOURCES mcgen_core SOURCES)
list(TRANSFORM MCGEN_SOURCES PREPEND "${MCGEN_SOURCE}/")
list(FILTER MCGEN_SOURCES EXCLUDE REGEX "/src/stats.cpp$")
file(READ "${MCGEN_SOURCE}/src/stats.cpp" MCGEN_STATS)
string(REPLACE "producer_late > 0.001" "producer_late > 0.01" MCGEN_STATS "${MCGEN_STATS}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/stats.cpp" "${MCGEN_STATS}")
list(APPEND MCGEN_SOURCES "${CMAKE_CURRENT_BINARY_DIR}/stats.cpp")
set_property(TARGET mcgen_core PROPERTY SOURCES "${MCGEN_SOURCES}")
set_target_properties(memcached-trace-generator PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
if(BUILD_TESTING)
  add_executable(test_mix_validity test_mix_validity.cpp)
  set_target_properties(test_mix_validity PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
  target_link_libraries(test_mix_validity PRIVATE mcgen_core)
  add_test(NAME test_mix_validity COMMAND test_mix_validity)
endif()
